// This file is NOT part of any Go module here: it is a template that
// scripts/strict-decode/strict_decode_fixtures.py copies, as
// zz_strict_decode_oracle_test.go, into the pinned go-algorand checkout's
// network package (the message types are unexported) and runs there.
// Issue #2396.
//
// For every {"name","kind","hex"} row of $STRICT_IN it runs go's real decode of
// that message and writes {"name","ok","err","enc"} to $STRICT_OUT (`enc`: the
// hex of go's re-encode of the decoded value, issue #2421):
//
//	idchal    identityChallengeSigned          protocol.Decode (netidentity.go,
//	          VerifyRequestAndAttachResponse, after the base64 header decode)
//	idresp    identityChallengeResponseSigned  protocol.Decode (VerifyResponse)
//	idverify  identityVerificationMessageSigned protocol.Decode
//	          (identityVerificationHandler)
//	peermeta  peerMetaHeaders                  UnmarshalMsg (p2pMetainfo.go,
//	          readPeerMetaHeaders)
//
// and, with $STRICT_SEEDS_OUT, writes fully populated random seeds, and, with
// $STRICT_SCHEMA_OUT, the msgp field layout of the three identity messages (keys
// in go's declaration order: the positional order of go's struct-from-array
// decode, issue #2426). `peerMetaHeaders` is a map type, not a struct.

func TestStrictDecodeOracleNetwork(t *testing.T) {
	in, err := os.Open(os.Getenv("STRICT_IN"))
	if err != nil {
		t.Fatal(err)
	}
	defer in.Close()
	out, err := os.Create(os.Getenv("STRICT_OUT"))
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()

	type row struct {
		Name string `json:"name"`
		Kind string `json:"kind"`
		Hex  string `json:"hex"`
	}
	type res struct {
		Name string `json:"name"`
		Ok   bool   `json:"ok"`
		Err  string `json:"err"`
		// go's canonical re-encode of the decoded value (accepted rows
		// only): pins what a repeated key decodes to (issue #2421)
		Enc string `json:"enc,omitempty"`
	}
	sc := bufio.NewScanner(in)
	sc.Buffer(make([]byte, 1<<20), 1<<28)
	enc := json.NewEncoder(out)
	for sc.Scan() {
		var r row
		if err := json.Unmarshal(sc.Bytes(), &r); err != nil {
			t.Fatal(err)
		}
		b, err := hex.DecodeString(r.Hex)
		if err != nil {
			t.Fatal(err)
		}
		func() {
			o := res{Name: r.Name}
			defer func() {
				if x := recover(); x != nil {
					o.Ok, o.Err = false, fmt.Sprintf("panic: %v", x)
				}
				enc.Encode(o)
			}()
			var derr error
			var val msgp.Marshaler
			switch r.Kind {
			case "idchal":
				var v identityChallengeSigned
				derr, val = protocol.Decode(b, &v), &v
			case "idresp":
				var v identityChallengeResponseSigned
				derr, val = protocol.Decode(b, &v), &v
			case "idverify":
				var v identityVerificationMessageSigned
				derr, val = protocol.Decode(b, &v), &v
			case "peermeta":
				var v peerMetaHeaders
				_, derr = v.UnmarshalMsg(b)
				// nil and empty are the same to every reader of the
				// headers (p2pMetainfo.go): encode both as empty
				if v == nil {
					v = peerMetaHeaders{}
				}
				for k := range v {
					if v[k] == nil {
						v[k] = peerMetaValues{}
					}
				}
				val = &v
			default:
				t.Fatalf("unknown kind %q", r.Kind)
			}
			o.Ok = derr == nil
			if derr != nil {
				o.Err = derr.Error()
			} else {
				o.Enc = hex.EncodeToString(val.MarshalMsg(nil))
			}
		}()
	}
}

// TestStrictDecodeSeedsNetwork writes $STRICT_SEEDS_OUT: one go-encoded, fully
// populated (protocol.RandomizeObject) value per kind and seed.
func TestStrictDecodeSeedsNetwork(t *testing.T) {
	path := os.Getenv("STRICT_SEEDS_OUT")
	if path == "" {
		t.Skip("STRICT_SEEDS_OUT not set")
	}
	out, err := os.Create(path)
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()
	enc := json.NewEncoder(out)
	rand.Seed(2396)
	emit := func(kind string, i int, tmpl any) {
		o, err := protocol.RandomizeObject(tmpl, protocol.RandomizeObjectWithMaxCollectionLen(2))
		if err != nil {
			t.Fatal(err)
		}
		m, ok := o.(msgp.Marshaler)
		if !ok {
			t.Fatalf("%T is not a msgp.Marshaler", o)
		}
		enc.Encode(map[string]string{"name": fmt.Sprintf("%s_rand%d", kind, i), "kind": kind, "hex": hex.EncodeToString(m.MarshalMsg(nil))})
	}
	for i := 0; i < 4; i++ {
		emit("idchal", i, &identityChallengeSigned{})
		emit("idresp", i, &identityChallengeResponseSigned{})
		emit("idverify", i, &identityVerificationMessageSigned{})
		emit("peermeta", i, &peerMetaHeaders{})
	}
}

// schemaType describes one field's value: a leaf (a scalar named by its Go
// kind, "bytes", "bytes[N]", or a struct with a hand-written msgp codec), a
// struct, or a list / map of those.
type schemaType struct {
	Kind string      `json:"k"`
	Name string      `json:"t,omitempty"`
	Elem *schemaType `json:"e,omitempty"`
}

type schemaField struct {
	GoName string     `json:"go"`
	Key    string     `json:"key"`
	Type   schemaType `json:"type"`
}

// opaqueStructs are structs whose msgp codec is written by hand (`//msgp:ignore`
// and custom marshalers): they are one value on the wire, not a map.
var opaqueStructs = map[string]bool{"basics.MicroAlgos": true}

func schemaOfType(rt reflect.Type, out map[string][]schemaField) schemaType {
	for rt.Kind() == reflect.Ptr {
		rt = rt.Elem()
	}
	switch rt.Kind() {
	case reflect.Struct:
		name := rt.String()
		if opaqueStructs[name] {
			return schemaType{Kind: "leaf", Name: "uint64"}
		}
		if _, done := out[name]; !done {
			out[name] = nil // cut recursion
			out[name] = schemaFields(rt, out)
		}
		return schemaType{Kind: "struct", Name: name}
	case reflect.Slice, reflect.Array:
		if rt.Elem().Kind() == reflect.Uint8 {
			if rt.Kind() == reflect.Array {
				return schemaType{Kind: "leaf", Name: fmt.Sprintf("bytes[%d]", rt.Len())}
			}
			return schemaType{Kind: "leaf", Name: "bytes"}
		}
		e := schemaOfType(rt.Elem(), out)
		return schemaType{Kind: "list", Elem: &e}
	case reflect.Map:
		e := schemaOfType(rt.Elem(), out)
		return schemaType{Kind: "map", Elem: &e}
	}
	// a scalar: its Go kind (uint64, bool, string, ...)
	return schemaType{Kind: "leaf", Name: rt.Kind().String()}
}

// schemaFields lists a struct's msgp fields in declaration order: an embedded
// struct without a codec name is flattened in place, a `codec:"-"` or
// unexported field is skipped, and an empty codec name is the Go field name.
func schemaFields(rt reflect.Type, out map[string][]schemaField) []schemaField {
	var fs []schemaField
	for i := 0; i < rt.NumField(); i++ {
		f := rt.Field(i)
		tag := f.Tag.Get("codec")
		name := strings.Split(tag, ",")[0]
		if name == "-" || f.Name == "_struct" || (f.PkgPath != "" && !f.Anonymous) {
			continue
		}
		if f.Anonymous && name == "" {
			ft := f.Type
			for ft.Kind() == reflect.Ptr {
				ft = ft.Elem()
			}
			if ft.Kind() == reflect.Struct {
				fs = append(fs, schemaFields(ft, out)...)
				continue
			}
		}
		if name == "" {
			name = f.Name
		}
		fs = append(fs, schemaField{GoName: f.Name, Key: name, Type: schemaOfType(f.Type, out)})
	}
	return fs
}

// TestStrictDecodeSchemaNetwork writes $STRICT_SCHEMA_OUT.
func TestStrictDecodeSchemaNetwork(t *testing.T) {
	path := os.Getenv("STRICT_SCHEMA_OUT")
	if path == "" {
		t.Skip("STRICT_SCHEMA_OUT not set")
	}
	out := map[string][]schemaField{}
	for _, v := range []any{
		identityChallengeSigned{}, identityChallengeResponseSigned{}, identityVerificationMessageSigned{},
	} {
		schemaOfType(reflect.TypeOf(v), out)
	}
	b, err := json.MarshalIndent(out, "", " ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
