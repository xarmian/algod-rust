// This file is NOT part of any Go module here: it is a template that
// scripts/strict-decode/strict_decode_fixtures.py copies, as
// zz_strict_decode_oracle_test.go, into the pinned go-algorand checkout's
// ledger package (the catchpoint types are unexported) and runs there.
// Issue #2396.
//
// For every {"name","kind","hex"} row of $STRICT_IN it runs what go's catchpoint
// restore does with that catchpoint file section (ledger/catchupaccessor.go)
// and writes {"name","ok","err","enc"} to $STRICT_OUT (`enc`: the hex of go's
// re-encode of the decoded value, issue #2421):
//
//	cpheader  content.msgpack: protocol.Decode into CatchpointFileHeader
//	          (processStagingContent)
//	cpchunk   balances.N.msgpack: protocol.Decode into CatchpointSnapshotChunkV6,
//	          then prepareNormalizedBalancesV6, which decodes every msgp.Raw
//	          account (`b`) and resource (`c`) blob (processStagingBalances)
//	cpspver   stateProofVerificationContext.msgpack: protocol.Decode into
//	          catchpointStateProofVerificationContext
//	          (processStagingStateProofVerificationContext)
//
// and, with $STRICT_SEEDS_OUT, writes fully populated random seeds; with
// $STRICT_SCHEMA_OUT (TestStrictDecodeSchemaLedger) the msgp field layout of the
// types above (issue #2609: go v5.1.1 decodes none of them from an array).

func TestStrictDecodeOracleLedger(t *testing.T) {
	in, err := os.Open(os.Getenv("STRICT_IN"))
	if err != nil {
		t.Fatal(err)
	}
	defer in.Close()
	out, err := os.Create(os.Getenv("STRICT_OUT"))
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()

	type row struct {
		Name string `json:"name"`
		Kind string `json:"kind"`
		Hex  string `json:"hex"`
	}
	type res struct {
		Name string `json:"name"`
		Ok   bool   `json:"ok"`
		Err  string `json:"err"`
		// go's canonical re-encode of the decoded value (accepted rows
		// only): pins what a repeated key decodes to (issue #2421)
		Enc string `json:"enc,omitempty"`
	}
	proto := config.Consensus[protocol.ConsensusCurrentVersion]
	sc := bufio.NewScanner(in)
	sc.Buffer(make([]byte, 1<<20), 1<<28)
	enc := json.NewEncoder(out)
	for sc.Scan() {
		var r row
		if err := json.Unmarshal(sc.Bytes(), &r); err != nil {
			t.Fatal(err)
		}
		b, err := hex.DecodeString(r.Hex)
		if err != nil {
			t.Fatal(err)
		}
		func() {
			o := res{Name: r.Name}
			defer func() {
				if x := recover(); x != nil {
					o.Ok, o.Err = false, fmt.Sprintf("panic: %v", x)
				}
				enc.Encode(o)
			}()
			var derr error
			var val []byte
			switch r.Kind {
			case "cpheader":
				var v CatchpointFileHeader
				if derr = protocol.Decode(b, &v); derr == nil {
					val = protocol.Encode(&v)
				}
			case "cpchunk":
				// the chunk, then every decoded account blob and every
				// decoded resource blob (ascending creatable index)
				var v CatchpointSnapshotChunkV6
				derr = protocol.Decode(b, &v)
				if derr == nil {
					var nb []trackerdb.NormalizedAccountBalance
					nb, derr = prepareNormalizedBalancesV6(v.Balances, proto)
					if derr == nil {
						val = protocol.Encode(&v)
						for i := range nb {
							val = append(val, protocol.Encode(&nb[i].AccountData)...)
							idx := make([]basics.CreatableIndex, 0, len(nb[i].Resources))
							for k := range nb[i].Resources {
								idx = append(idx, k)
							}
							sort.Slice(idx, func(a, c int) bool { return idx[a] < idx[c] })
							for _, k := range idx {
								rd := nb[i].Resources[k]
								val = append(val, protocol.Encode(&rd)...)
							}
						}
					}
				}
			case "cpspver":
				var v catchpointStateProofVerificationContext
				if derr = protocol.Decode(b, &v); derr == nil {
					val = protocol.Encode(&v)
				}
			default:
				t.Fatalf("unknown kind %q", r.Kind)
			}
			o.Ok = derr == nil
			if derr != nil {
				o.Err = derr.Error()
			} else {
				o.Enc = hex.EncodeToString(val)
			}
		}()
	}
}

// TestStrictDecodeSeedsLedger writes $STRICT_SEEDS_OUT: go-encoded, fully
// populated (protocol.RandomizeObject) values per kind. A chunk's msgp.Raw
// fields hold the encoding of a random value of the type go decodes them into
// (or, for `oa` / `orp`, stores), as the catchpoint writer produces them.
func TestStrictDecodeSeedsLedger(t *testing.T) {
	path := os.Getenv("STRICT_SEEDS_OUT")
	if path == "" {
		t.Skip("STRICT_SEEDS_OUT not set")
	}
	out, err := os.Create(path)
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()
	enc := json.NewEncoder(out)
	rand.Seed(2396)
	random := func(tmpl any) msgp.Marshaler {
		o, err := protocol.RandomizeObject(tmpl, protocol.RandomizeObjectWithMaxCollectionLen(2))
		if err != nil {
			t.Fatal(err)
		}
		m, ok := o.(msgp.Marshaler)
		if !ok {
			t.Fatalf("%T is not a msgp.Marshaler", o)
		}
		return m
	}
	emit := func(kind string, i int, m msgp.Marshaler) {
		enc.Encode(map[string]string{"name": fmt.Sprintf("%s_rand%d", kind, i), "kind": kind, "hex": hex.EncodeToString(protocol.Encode(m))})
	}
	for i := 0; i < 4; i++ {
		emit("cpheader", i, random(&CatchpointFileHeader{}))
		emit("cpspver", i, random(&catchpointStateProofVerificationContext{}))

		var chunk CatchpointSnapshotChunkV6
		for j := 0; j < 2; j++ {
			var addr basics.Address
			rand.Read(addr[:])
			bal := encoded.BalanceRecordV6{
				Address:              addr,
				AccountData:          protocol.Encode(random(&trackerdb.BaseAccountData{})),
				Resources:            map[uint64]msgp.Raw{},
				ExpectingMoreEntries: j == 0,
			}
			for k := 0; k < 2; k++ {
				bal.Resources[uint64(rand.Int63())] = protocol.Encode(random(&trackerdb.ResourcesData{}))
			}
			chunk.Balances = append(chunk.Balances, bal)
			chunk.KVs = append(chunk.KVs, *random(&encoded.KVRecordV6{}).(*encoded.KVRecordV6))
			// RandomizeObject refuses msgp.Raw fields: fill these by hand
			oa := encoded.OnlineAccountRecordV6{
				UpdateRound:             basics.Round(rand.Uint64()),
				NormalizedOnlineBalance: rand.Uint64(),
				VoteLastValid:           basics.Round(rand.Uint64()),
				Data:                    protocol.Encode(random(&trackerdb.BaseOnlineAccountData{})),
			}
			rand.Read(oa.Address[:])
			chunk.OnlineAccounts = append(chunk.OnlineAccounts, oa)
			chunk.OnlineRoundParams = append(chunk.OnlineRoundParams, encoded.OnlineRoundParamsRecordV6{
				Round: basics.Round(rand.Uint64()),
				Data:  protocol.Encode(random(&ledgercore.OnlineRoundParamsData{})),
			})
		}
		emit("cpchunk", i, &chunk)
	}
}

// schemaType describes one field's value: a leaf (a scalar named by its Go
// kind, "bytes", "bytes[N]", "raw" for a msgp.Raw go keeps undecoded), a
// struct, or a list / map of those.
type schemaType struct {
	Kind string      `json:"k"`
	Name string      `json:"t,omitempty"`
	Elem *schemaType `json:"e,omitempty"`
}

type schemaField struct {
	GoName string     `json:"go"`
	Key    string     `json:"key"`
	Type   schemaType `json:"type"`
}

// opaqueStructs are structs whose msgp codec is written by hand: they are one
// value on the wire, not a map.
var opaqueStructs = map[string]bool{"basics.MicroAlgos": true}

// rawTypes: the type go decodes the msgp.Raw field of a struct into (the
// catchpoint restore decodes `b` and `c` in prepareNormalizedBalancesV6; the
// other msgp.Raw fields stay undecoded and are "raw" leaves).
var rawTypes = map[string]schemaType{
	"encoded.BalanceRecordV6.b": {Kind: "struct", Name: "trackerdb.BaseAccountData"},
	"encoded.BalanceRecordV6.c": {Kind: "map", Elem: &schemaType{Kind: "struct", Name: "trackerdb.ResourcesData"}},
}

func schemaOfType(rt reflect.Type, out map[string][]schemaField) schemaType {
	for rt.Kind() == reflect.Ptr {
		rt = rt.Elem()
	}
	if rt == reflect.TypeOf(msgp.Raw{}) {
		return schemaType{Kind: "leaf", Name: "raw"}
	}
	switch rt.Kind() {
	case reflect.Struct:
		name := rt.String()
		if opaqueStructs[name] {
			return schemaType{Kind: "leaf", Name: "uint64"}
		}
		if _, done := out[name]; !done {
			out[name] = nil // cut recursion
			out[name] = schemaFields(rt, out)
		}
		return schemaType{Kind: "struct", Name: name}
	case reflect.Slice, reflect.Array:
		if rt.Elem().Kind() == reflect.Uint8 {
			if rt.Kind() == reflect.Array {
				return schemaType{Kind: "leaf", Name: fmt.Sprintf("bytes[%d]", rt.Len())}
			}
			return schemaType{Kind: "leaf", Name: "bytes"}
		}
		e := schemaOfType(rt.Elem(), out)
		return schemaType{Kind: "list", Elem: &e}
	case reflect.Map:
		e := schemaOfType(rt.Elem(), out)
		return schemaType{Kind: "map", Elem: &e}
	}
	return schemaType{Kind: "leaf", Name: rt.Kind().String()}
}

// schemaFields lists a struct's msgp fields in declaration order: an embedded
// struct without a codec name is flattened in place, a `codec:"-"` or
// unexported field is skipped, and an empty codec name is the Go field name.
func schemaFields(rt reflect.Type, out map[string][]schemaField) []schemaField {
	var fs []schemaField
	for i := 0; i < rt.NumField(); i++ {
		f := rt.Field(i)
		tag := f.Tag.Get("codec")
		name := strings.Split(tag, ",")[0]
		if name == "-" || f.Name == "_struct" || (f.PkgPath != "" && !f.Anonymous) {
			continue
		}
		if f.Anonymous && name == "" {
			ft := f.Type
			for ft.Kind() == reflect.Ptr {
				ft = ft.Elem()
			}
			if ft.Kind() == reflect.Struct {
				fs = append(fs, schemaFields(ft, out)...)
				continue
			}
		}
		if name == "" {
			name = f.Name
		}
		st := schemaOfType(f.Type, out)
		if o, ok := rawTypes[rt.String()+"."+name]; ok {
			st = o
			if o.Kind == "struct" {
				schemaOfType(reflect.TypeOf(trackerdb.BaseAccountData{}), out)
			} else {
				schemaOfType(reflect.TypeOf(trackerdb.ResourcesData{}), out)
			}
		}
		fs = append(fs, schemaField{GoName: f.Name, Key: name, Type: st})
	}
	return fs
}

// TestStrictDecodeSchemaLedger writes $STRICT_SCHEMA_OUT.
func TestStrictDecodeSchemaLedger(t *testing.T) {
	path := os.Getenv("STRICT_SCHEMA_OUT")
	if path == "" {
		t.Skip("STRICT_SCHEMA_OUT not set")
	}
	out := map[string][]schemaField{}
	for _, v := range []any{
		CatchpointFileHeader{}, CatchpointSnapshotChunkV6{}, catchpointStateProofVerificationContext{},
	} {
		schemaOfType(reflect.TypeOf(v), out)
	}
	b, err := json.MarshalIndent(out, "", " ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, append(b, '\n'), 0o644); err != nil {
		t.Fatal(err)
	}
}
