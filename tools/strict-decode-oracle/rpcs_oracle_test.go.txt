// This file is NOT part of any Go module here: it is a template that
// scripts/strict-decode/strict_decode_fixtures.py copies, as
// zz_strict_decode_oracle_test.go, into the pinned go-algorand checkout's
// rpcs package and runs there. Issue #2394.
//
// For every {"name","kind","hex"} row of $STRICT_IN it runs go's real
// protocol.Decode of that type and writes {"name","ok","err","enc"} to
// $STRICT_OUT (`enc`: the hex of go's re-encode of the decoded value, #2421):
//
//	blockcert    rpcs.EncodedBlockCert     (catchup / REST msgpack blocks)
//	block        bookkeeping.Block         (block DB, proposal block part)
//	header       bookkeeping.BlockHeader
//	stxn         transactions.SignedTxn    (TX gossip, REST submit)
//	stxninblock  transactions.SignedTxnInBlock
//	txn          transactions.Transaction
//	applydata    transactions.ApplyData
//	evaldelta    transactions.EvalDelta
//	txsync       []transactions.SignedTxn  (tx-sync response, rpcs/httpTxSync.go:
//	                                        protocol.DecodeReflect, issue #2396)
//
// and, with $STRICT_SEEDS_OUT, writes fully populated random seeds.

package rpcs

import (

	"encoding/hex"
	"encoding/json"
	"fmt"
	"math/rand"
	"os"
	"testing"

	"github.com/algorand/go-algorand/data/bookkeeping"

	"github.com/algorand/msgp/msgp"
)

func TestStrictDecodeOracleRpcs(t *testing.T) {
	in, err := os.Open(os.Getenv("STRICT_IN"))
	if err != nil {
		t.Fatal(err)
	}
	defer in.Close()
	out, err := os.Create(os.Getenv("STRICT_OUT"))
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()

	type row struct {
		Name string `json:"name"`
		Kind string `json:"kind"`
		Hex  string `json:"hex"`
	}
	type res struct {
		Name string `json:"name"`
		Ok   bool   `json:"ok"`
		Err  string `json:"err"`
		// go's canonical re-encode of the decoded value (accepted rows
		// only): pins what a repeated key decodes to (issue #2421)
		Enc string `json:"enc,omitempty"`
	}
	sc := bufio.NewScanner(in)
	sc.Buffer(make([]byte, 1<<20), 1<<28)
	enc := json.NewEncoder(out)
	for sc.Scan() {
		var r row
		if err := json.Unmarshal(sc.Bytes(), &r); err != nil {
			t.Fatal(err)
		}
		b, err := hex.DecodeString(r.Hex)
		if err != nil {
			t.Fatal(err)
		}
		func() {
			o := res{Name: r.Name}
			defer func() {
				if x := recover(); x != nil {
					o.Ok, o.Err = false, fmt.Sprintf("panic: %v", x)
				}
				enc.Encode(o)
			}()
			var derr error
			var val []byte
			dec := func(m msgp.Unmarshaler) {
				if derr = protocol.Decode(b, m); derr == nil {
					val = protocol.Encode(m.(msgp.Marshaler))
				}
			}
			switch r.Kind {
			case "blockcert":
				dec(&EncodedBlockCert{})
			case "block":
				dec(&bookkeeping.Block{})
			case "header":
				dec(&bookkeeping.BlockHeader{})
			case "stxn":
				dec(&transactions.SignedTxn{})
			case "stxninblock":
				dec(&transactions.SignedTxnInBlock{})
			case "txn":
				dec(&transactions.Transaction{})
			case "applydata":
				dec(&transactions.ApplyData{})
			case "evaldelta":
				dec(&transactions.EvalDelta{})
			case "txsync":
				// the go-codec decode, then each transaction's msgp encode
				var v []transactions.SignedTxn
				if derr = protocol.DecodeReflect(b, &v); derr == nil {
					val = []byte{}
					for i := range v {
						val = append(val, protocol.Encode(&v[i])...)
					}
				}
			default:
				t.Fatalf("unknown kind %q", r.Kind)
			}
			o.Ok = derr == nil
			if derr != nil {
				o.Err = derr.Error()
			} else {
				o.Enc = hex.EncodeToString(val)
			}
		}()
	}
}

// TestStrictDecodeSeedsRpcs writes $STRICT_SEEDS_OUT: one go-encoded, fully
// populated (protocol.RandomizeObject) value per kind and seed.
func TestStrictDecodeSeedsRpcs(t *testing.T) {
	path := os.Getenv("STRICT_SEEDS_OUT")
	if path == "" {
		t.Skip("STRICT_SEEDS_OUT not set")
	}
	out, err := os.Create(path)
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()
	enc := json.NewEncoder(out)
	n := 4
	rand.Seed(2394)
	emit := func(kind string, i int, tmpl any) {
		o, err := protocol.RandomizeObject(tmpl, protocol.RandomizeObjectWithMaxCollectionLen(2))
		if err != nil {
			t.Fatal(err)
		}
		m, ok := o.(msgp.Marshaler)
		if !ok {
			t.Fatalf("%T is not a msgp.Marshaler", o)
		}
		enc.Encode(map[string]string{"name": fmt.Sprintf("%s_rand%d", kind, i), "kind": kind, "hex": hex.EncodeToString(protocol.Encode(m))})
	}
	for i := 0; i < n; i++ {
		emit("blockcert", i, &EncodedBlockCert{})
		emit("block", i, &bookkeeping.Block{})
		emit("header", i, &bookkeeping.BlockHeader{})
		emit("stxn", i, &transactions.SignedTxn{})
		emit("stxninblock", i, &transactions.SignedTxnInBlock{})
		emit("txn", i, &transactions.Transaction{})
		emit("applydata", i, &transactions.ApplyData{})
		emit("evaldelta", i, &transactions.EvalDelta{})
	}
}

// TestStrictDecodeSeedsTxSync writes $STRICT_SEEDS_OUT: tx-sync responses
// (rpcs/txService.go encodes them with protocol.EncodeReflect) of fully
// populated random signed transactions. Issue #2396.
func TestStrictDecodeSeedsTxSync(t *testing.T) {
	path := os.Getenv("STRICT_SEEDS_OUT")
	if path == "" {
		t.Skip("STRICT_SEEDS_OUT not set")
	}
	out, err := os.Create(path)
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()
	enc := json.NewEncoder(out)
	rand.Seed(2396)
	for i := 0; i < 4; i++ {
		var txns []transactions.SignedTxn
		for j := 0; j < 1+i%2; j++ {
			o, err := protocol.RandomizeObject(&transactions.SignedTxn{}, protocol.RandomizeObjectWithMaxCollectionLen(2))
			if err != nil {
				t.Fatal(err)
			}
			txns = append(txns, *o.(*transactions.SignedTxn))
		}
		enc.Encode(map[string]string{"name": fmt.Sprintf("txsync_rand%d", i), "kind": "txsync", "hex": hex.EncodeToString(protocol.EncodeReflect(txns))})
	}
}
