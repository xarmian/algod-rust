// This file is NOT part of any Go module here: it is a template that
// scripts/strict-decode/strict_decode_fixtures.py copies, as
// zz_strict_decode_oracle_test.go, into the pinned go-algorand checkout's
// agreement package (the payload/vote/bundle decoders are unexported) and runs
// there. Issue #2394.
//
// For every {"name","kind","hex"} row of $STRICT_IN it runs go's real network
// decode of that message kind and writes {"name","ok","err","enc"} to
// $STRICT_OUT (`enc`: the hex of go's re-encode of the decoded value, #2421):
//
//	payload  protocol.Decode into transmittedPayload      (decodeProposal, the "PP" handler)
//	vote     protocol.Decode into unauthenticatedVote     (decodeVote, the "AV" handler)
//	bundle   protocol.Decode into unauthenticatedBundle   (decodeBundle, the "VB" handler)
//	cert     protocol.Decode into Certificate             (the block DB's cert)
//
// The handlers' wellFormed() checks run after the decode and are not part of it.

package agreement

import (
	"bufio"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"math/rand"
	"os"
	"testing"

	"github.com/algorand/go-algorand/protocol"

func TestStrictDecodeOracleAgreement(t *testing.T) {
	in, err := os.Open(os.Getenv("STRICT_IN"))
	if err != nil {
		t.Fatal(err)
	}
	defer in.Close()
	out, err := os.Create(os.Getenv("STRICT_OUT"))
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()

	type row struct {
		Name string `json:"name"`
		Kind string `json:"kind"`
		Hex  string `json:"hex"`
	}
	type res struct {
		Name string `json:"name"`
		Ok   bool   `json:"ok"`
		Err  string `json:"err"`
		// go's canonical re-encode of the decoded value (accepted rows
		// only): pins what a repeated key decodes to (issue #2421)
		Enc string `json:"enc,omitempty"`
	}
	sc := bufio.NewScanner(in)
	sc.Buffer(make([]byte, 1<<20), 1<<28)
	enc := json.NewEncoder(out)
	for sc.Scan() {
		var r row
		if err := json.Unmarshal(sc.Bytes(), &r); err != nil {
			t.Fatal(err)
		}
		b, err := hex.DecodeString(r.Hex)
		if err != nil {
			t.Fatal(err)
		}
		func() {
			o := res{Name: r.Name}
			defer func() {
				if x := recover(); x != nil {
					o.Ok, o.Err = false, fmt.Sprintf("panic: %v", x)
				}
				enc.Encode(o)
			}()
			var derr error
			var val []byte
			switch r.Kind {
			case "payload":
				var v transmittedPayload
				if derr = protocol.Decode(b, &v); derr == nil {
					val = protocol.Encode(&v)
				}
			case "vote":
				var v unauthenticatedVote
				if derr = protocol.Decode(b, &v); derr == nil {
					val = protocol.Encode(&v)
				}
			case "bundle":
				var v unauthenticatedBundle
				if derr = protocol.Decode(b, &v); derr == nil {
					val = protocol.Encode(&v)
				}
			case "cert":
				var c Certificate
				if derr = protocol.Decode(b, &c); derr == nil {
					val = protocol.Encode(&c)
				}
			default:
				t.Fatalf("unknown kind %q", r.Kind)
			}
			o.Ok = derr == nil
			if derr != nil {
				o.Err = derr.Error()
			} else {
				o.Enc = hex.EncodeToString(val)
			}
		}()
	}
}

// TestStrictDecodeSeedsAgreement writes $STRICT_SEEDS_OUT: one go-encoded,
// fully populated (protocol.RandomizeObject) message per kind and seed.
func TestStrictDecodeSeedsAgreement(t *testing.T) {
	path := os.Getenv("STRICT_SEEDS_OUT")
	if path == "" {
		t.Skip("STRICT_SEEDS_OUT not set")
	}
	out, err := os.Create(path)
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()
	enc := json.NewEncoder(out)
	n := 4
	rand.Seed(2394)
	emit := func(kind string, i int, tmpl any) {
		o, err := protocol.RandomizeObject(tmpl, protocol.RandomizeObjectWithMaxCollectionLen(2))
		if err != nil {
			t.Fatal(err)
		}
		m, ok := o.(msgp.Marshaler)
		if !ok {
			t.Fatalf("%T is not a msgp.Marshaler", o)
		}
		enc.Encode(map[string]string{"name": fmt.Sprintf("%s_rand%d", kind, i), "kind": kind, "hex": hex.EncodeToString(protocol.Encode(m))})
	}
	for i := 0; i < n; i++ {
		emit("payload", i, &transmittedPayload{})
		emit("vote", i, &unauthenticatedVote{})
		emit("bundle", i, &unauthenticatedBundle{})
		emit("cert", i, &Certificate{})
	}
}
